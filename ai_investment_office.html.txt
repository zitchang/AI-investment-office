        function createVoxel(w, h, d, color, x = 0, y = 0, z = 0, castShadow = true, receiveShadow = true) {
            const geo = new THREE.BoxGeometry(w, h, d);
            const mat = new THREE.MeshStandardMaterial({
                color: color,
                roughness: 0.72,
                metalness: 0.1
            });
            const mesh = new THREE.Mesh(geo, mat);
            mesh.position.set(x, y, z);
            mesh.castShadow = castShadow;
            mesh.receiveShadow = receiveShadow;
            return mesh;
        }

        function createMonitorTexture(mode = 'kline') {
            const canvas = document.createElement('canvas');
            canvas.width = 128;
            canvas.height = 80;
            const ctx = canvas.getContext('2d');

            ctx.fillStyle = '#070B12';
            ctx.fillRect(0, 0, 128, 80);

            ctx.strokeStyle = '#1E293B';
            ctx.lineWidth = 1;
            for (let y = 16; y < 80; y += 16) {
                ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(128, y); ctx.stroke();
            }

            if (mode === 'kline') {
                const candles = [
                    { x: 10, o: 56, c: 46, h: 42, l: 60, up: true },
                    { x: 22, o: 46, c: 50, h: 44, l: 54, up: false },
                    { x: 34, o: 50, c: 38, h: 34, l: 52, up: true },
                    { x: 46, o: 38, c: 32, h: 28, l: 42, up: true },
                    { x: 58, o: 32, c: 37, h: 30, l: 40, up: false },
                    { x: 70, o: 37, c: 24, h: 20, l: 38, up: true },
                    { x: 82, o: 24, c: 28, h: 22, l: 32, up: false },
                    { x: 94, o: 28, c: 15, h: 11, l: 30, up: true },
                    { x: 106, o: 15, c: 11, h: 8, l: 19, up: true }
                ];
                candles.forEach(c => {
                    ctx.strokeStyle = c.up ? '#EF4444' : '#10B981';
                    ctx.fillStyle = c.up ? '#EF4444' : '#10B981';
                    ctx.beginPath();
                    ctx.moveTo(c.x + 3, c.h);
                    ctx.lineTo(c.x + 3, c.l);
                    ctx.stroke();
                    ctx.fillRect(c.x, Math.min(c.o, c.c), 6, Math.max(3, Math.abs(c.o - c.c)));
                });
            } else {
                const bars = ['#3B82F6', '#F59E0B', '#10B981', '#EF4444', '#8B5CF6'];
                bars.forEach((col, i) => {
                    ctx.fillStyle = col;
                    ctx.fillRect(10, 12 + i * 13, 40 + ((i * 23) % 65), 6);
                });
            }
            const tex = new THREE.CanvasTexture(canvas);
            tex.magFilter = THREE.NearestFilter;
            return tex;
        }

        function createWallBoardTexture(text, subText, accentColor) {
            const canvas = document.createElement('canvas');
            canvas.width = 320;

            ctx.fillStyle = '#0B0F17';
            ctx.fillRect(0, 0, 320, 80);

        function initThreeScene() {
            const container = document.getElementById('three-container');
            scene = new THREE.Scene();
            scene.background = new THREE.Color('#0B0F17');

            camera = new THREE.PerspectiveCamera(34, container.clientWidth / container.clientHeight, 0.1, 100);
            updateCameraPosition();

            renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'high-performance' });
            renderer.setSize(container.clientWidth, container.clientHeight);
            renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
            renderer.shadowMap.enabled = true;
            renderer.shadowMap.type = THREE.PCFSoftShadowMap;
            container.appendChild(renderer.domElement);

            // Lighting
            const ambient = new THREE.AmbientLight(0xCBD5E1, 0.65);
            scene.add(ambient);

            const dirLight = new THREE.DirectionalLight(0xFFFBEB, 0.95);
            dirLight.position.set(11, 18, 12);
            dirLight.castShadow = true;
            dirLight.shadow.mapSize.width = 2048;
            dirLight.shadow.mapSize.height = 2048;
            const d = 11;
            dirLight.shadow.camera.left = -d;
            dirLight.shadow.camera.right = d;
            dirLight.shadow.camera.top = d;
            dirLight.shadow.camera.bottom = -d;
            dirLight.shadow.bias = -0.0005;
            scene.add(dirLight);

            const blueFill = new THREE.DirectionalLight(0x3B82F6, 0.35);
            blueFill.position.set(-10, 8, -10);
            scene.add(blueFill);

            officeGroup = new THREE.Group();
            scene.add(officeGroup);

            buildInstitutionalOffice();
            buildInstitutionalAgents();
            setupCameraControls();

            window.addEventListener('resize', () => {
                if (!container) return;
                camera.aspect = container.clientWidth / container.clientHeight;
                camera.updateProjectionMatrix();
                renderer.setSize(container.clientWidth, container.clientHeight);
            });

            animate();
        }

                    legR,

                    dotEl: document.getElementById(`dot-${member.id}`),
                    mode: 'idle', // 'idle' | 'to_desk' | 'working' | 'debating' | 'returning'

        function updateCameraPosition() {
            const x = camState.target.x + camState.distance * Math.sin(camState.polar) * Math.sin(camState.azimuth);
            const y = camState.target.y + camState.distance * Math.cos(camState.polar);
            const z = camState.target.z + camState.distance * Math.sin(camState.polar) * Math.cos(camState.azimuth);
            camera.position.set(x, y, z);
            camera.lookAt(camState.target);
        }

        function setupCameraControls() {
            const viewport = document.getElementById('office-viewport');

            viewport.addEventListener('mousedown', (e) => {
                camState.isDragging = true;
                camState.prevX = e.clientX;
                camState.prevY = e.clientY;
            });

            window.addEventListener('mousemove', (e) => {
                if (!camState.isDragging) return;
                const dx = e.clientX - camState.prevX;
                const dy = e.clientY - camState.prevY;
                camState.prevX = e.clientX;
                camState.prevY = e.clientY;

                camState.azimuth -= dx * 0.006;
                camState.polar = Math.max(0.35, Math.min(Math.PI / 2.2, camState.polar - dy * 0.006));
                updateCameraPosition();
            });

            window.addEventListener('mouseup', () => {
                camState.isDragging = false;
            });

            viewport.addEventListener('touchstart', (e) => {
                if (e.touches.length === 1) {
                    camState.isDragging = true;
                    camState.prevX = e.touches[0].clientX;
                    camState.prevY = e.touches[0].clientY;
                }
            }, { passive: true });

            viewport.addEventListener('touchmove', (e) => {
                if (!camState.isDragging || e.touches.length !== 1) return;
                const dx = e.touches[0].clientX - camState.prevX;
                const dy = e.touches[0].clientY - camState.prevY;
                camState.prevX = e.touches[0].clientX;
                camState.prevY = e.touches[0].clientY;

                camState.azimuth -= dx * 0.007;
                camState.polar = Math.max(0.35, Math.min(Math.PI / 2.2, camState.polar - dy * 0.007));
                updateCameraPosition();
            }, { passive: true });

            viewport.addEventListener('touchend', () => {
                camState.isDragging = false;
            });

            viewport.addEventListener('wheel', (e) => {
                e.preventDefault();
                camState.distance = Math.max(11, Math.min(28, camState.distance + e.deltaY * 0.01));
                updateCameraPosition();
            }, { passive: false });
        }

        const clock = new THREE.Clock();

        function animate() {
            requestAnimationFrame(animate);
            const delta = Math.min(clock.getDelta(), 0.1);
            const elapsed = clock.getElapsedTime();

            Object.values(agentEntities).forEach((agent, idx) => {
                const mesh = agent.mesh;

                if (agent.mode === 'to_desk' || agent.mode === 'returning') {
                    const dir = new THREE.Vector3().subVectors(agent.targetPos, mesh.position);
                    dir.y = 0;
                    const dist = dir.length();

                    if (dist > 0.08) {
                        dir.normalize();
                        mesh.position.addScaledVector(dir, 3.8 * delta);
                        mesh.rotation.y = Math.atan2(dir.x, dir.z);

                        // Brisk running animation
                        mesh.position.y = Math.abs(Math.sin(elapsed * 16 + idx)) * 0.14;
                        agent.legL.rotation.x = Math.sin(elapsed * 16 + idx) * 0.65;
                        agent.legR.rotation.x = -Math.sin(elapsed * 16 + idx) * 0.65;
                        agent.armL.rotation.x = -Math.sin(elapsed * 16 + idx) * 0.55;
                        agent.armR.rotation.x = Math.sin(elapsed * 16 + idx) * 0.55;
                    } else {
                        mesh.position.copy(agent.targetPos);
                        agent.legL.rotation.x = 0;
                        agent.legR.rotation.x = 0;

                        if (agent.mode === 'to_desk') {
                            agent.mode = 'working';
                            mesh.rotation.y = Math.PI;
                            mesh.position.y = 0.08;
                        } else {
                            agent.mode = 'idle';
                            mesh.rotation.y = agent.config.loungeRot;
                            mesh.position.y = 0.15;
                            agent.armL.rotation.x = -0.25;
                            agent.armR.rotation.x = -0.25;
                            setAgentDot(agent.config.id, 'done');
                            setTimeout(() => {
                                if (agent.mode === 'idle') setAgentDot(agent.config.id, 'idle');
                            }, 5000);
                        }
                    }
                } else if (agent.mode === 'working') {
                    // Rapid terminal typing
                    agent.armL.rotation.x = -0.88 + Math.sin(elapsed * 28 + idx) * 0.25;
                    agent.armR.rotation.x = -0.88 - Math.sin(elapsed * 28 + idx) * 0.25;
                    agent.head.rotation.y = Math.sin(elapsed * 4 + idx) * 0.2;
                } else if (agent.mode === 'debating') {
                    // Turn toward each other and gesture animatedly during IC Debate!
                    if (agent.config.id === 'elena') {
                        mesh.rotation.y = -Math.PI / 2.2; // Face Sophia & Marcus
                    } else if (agent.config.id === 'victor') {
                        mesh.rotation.y = Math.PI * 0.85;
                    } else {
                        mesh.rotation.y = Math.PI / 2.2; // Face Elena
                    }
                    agent.armR.rotation.x = -0.7 + Math.sin(elapsed * 12 + idx) * 0.45;
                    agent.head.rotation.y = Math.sin(elapsed * 6 + idx) * 0.25;
                } else {
                    // Idle in lounge
                    agent.head.rotation.y = Math.sin(elapsed * 1.3 + idx * 1.5) * 0.2;
                    agent.armL.rotation.x = -0.25;
                    agent.armR.rotation.x = -0.25;

                    agent.quoteTimer += delta;
                    if (agent.quoteTimer > 8.0) {
                        agent.quoteTimer = 0;
                        const qList = agent.config.idleQuotes;
                        setAgentBubble(agent.config.id, qList[Math.floor(Math.random() * qList.length)], false);
                    }
                }
            });

            renderer.render(scene, camera);
            projectLabelsTo2D();
        }

        function projectLabelsTo2D() {
            const container = document.getElementById('three-container');
            const w = container.clientWidth;
            const h = container.clientHeight;

            Object.values(agentEntities).forEach(agent => {
                const pos = new THREE.Vector3();
                agent.mesh.getWorldPosition(pos);
                pos.y += 2.18;
                pos.project(camera);

                agent.anchorEl.style.left = `${(pos.x * 0.5 + 0.5) * w}px`;
                agent.anchorEl.style.top = `${(-(pos.y * 0.5) + 0.5) * h}px`;
            });
        }

        async function runInvestmentCommittee(topic) {
            const cleanTopic = topic.trim();
            if (!cleanTopic) {
                showToast('請輸入欲研究的台美股代號、ETF 或總經議題', '⚠️');
                return;
            }
            if (isCommitteeRunning) {
                showToast('投審會正在進行中，請稍候本輪決議完成', '⏳');
                return;
            }

            isCommitteeRunning = true;
            const btnStart = document.getElementById('btn-start-ic');
            btnStart.disabled = true;
            btnStart.classList.add('opacity-60', 'cursor-not-allowed');

            // ----------------------------------------------------------------
            // STAGE 1: Data Gathering & Initial Draft (Run to Desks + API Call)
            // ----------------------------------------------------------------
            updateHeaderStatus('聯網研究中', 'text-amber-400', 'bg-amber-400 animate-ping', `投審會啟動：「${cleanTopic}」— 全員前往交易桌調閱數據`);
            updateStageBanner(true, '階段一：聯網數據搜集與初稿撰寫', 'STEP 1 / 4', 28, 'Sophia 正在檢索毛利率與估值倍數，Marcus 正在比對美債殖利率...');

            IC_TEAM.forEach(m => {
                const ag = agentEntities[m.id];
                ag.mode = 'to_desk';
                ag.targetPos.set(m.deskPos.x, 0, m.deskPos.z);
                setAgentDot(m.id, 'working');
            });

            setAgentBubble('victor', `投審會召集！全員針對「${cleanTopic.slice(0, 14)}...」建立量化模型`);
            setAgentBubble('sophia', `調閱「${cleanTopic.slice(0, 12)}...」營收、毛利率與 Forward PE 區間`);
            setAgentBubble('marcus', `檢索 Fed 利率點陣圖、10Y 美債殖利率與外資匯率流向`);
            setAgentBubble('elena', `掃描地緣關稅清單與供應鏈潛在斷鏈風險參數`);

            // Kick off real Gemini API call in parallel with stage animations
            const apiPromise = executeGeminiICAnalysis(cleanTopic);

            await new Promise(r => setTimeout(r, 2600));

            // ----------------------------------------------------------------
            // STAGE 2: CRO Stress Test & Bull/Bear Debate
            // ----------------------------------------------------------------
            updateHeaderStatus('投審會辯論中', 'text-rose-400', 'bg-rose-500 animate-ping', `CRO Elena 發起反方壓力測試，質疑樂觀估值與黑天鵝風險`);
            updateStageBanner(true, '階段二：CRO 壓力測試與多空辯論', 'STEP 2 / 4', 62, 'Elena 擔任反方辯手，針對估值過高與政策風險提出嚴厲反駁...');

            Object.values(agentEntities).forEach(ag => {
                ag.mode = 'debating';
                setAgentDot(ag.config.id, 'debate');
            });

            setAgentBubble('elena', `⚠️ 反方質疑：市場共識過度樂觀！若關稅或財報不如預期，估值恐回撤 15%！`, true);
            setAgentBubble('sophia', `基本面護城河仍在，但同意短線乖離偏大，建議改採分批承接！`, false);
            setAgentBubble('marcus', `實質利率尚未大幅下行，估值倍數確實不宜給予無上限溢價。`, false);
            setAgentBubble('victor', `雙方論點已收斂，我來計算最終風險報酬比與停損水位。`, false);

            await new Promise(r => setTimeout(r, 2400));

            // ----------------------------------------------------------------
            // STAGE 3: CIO Final Decision & Memorandum Synthesis
            // ----------------------------------------------------------------
            updateHeaderStatus('CIO 拍板定案', 'text-emerald-400', 'bg-emerald-400 animate-pulse', `CIO Victor 正在核定最終評級、資金配置比例與硬性停損點`);
            updateStageBanner(true, '階段三：CIO 拍板定案與備忘錄簽署', 'STEP 3 / 4', 88, 'Victor 正在彙整四方報告並產出機構級投資決策文件...');

            setAgentBubble('victor', `✅ 決議定案！已核定目標區間、風險報酬比與硬性停損紀律。`);

            // Wait for Gemini API response to complete
            const analysisData = await apiPromise;

            // ----------------------------------------------------------------
            // STAGE 4: Return to Lounge & Open Institutional Report Modal
            // ----------------------------------------------------------------
            const now = new Date();
            const timeStr = `${now.getFullYear()}/${String(now.getMonth() + 1).padStart(2, '0')}/${String(now.getDate()).padStart(2, '0')} ${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`;

                id: `ic-memo-${Date.now()}`,

                timestamp: timeStr,
                ...analysisData
            };

            });

            setTimeout(() => updateStageBanner(false), 1800);

            isCommitteeRunning = false;
            btnStart.disabled = false;
            btnStart.classList.remove('opacity-60', 'cursor-not-allowed');

            openICReportModal(finalReport);
        }

        function refreshApiStatusIndicators() {
            const indicator = document.getElementById('api-key-indicator');
            const modeBadge = document.getElementById('engine-mode-badge');
            const groundingTag = document.getElementById('grounding-status-tag');

            if (apiConfig.apiKey) {
                indicator.className = 'w-2 h-2 rounded-full bg-emerald-400 shadow-[0_0_6px_#10B981]';
                modeBadge.innerHTML = `<span class="w-1.5 h-1.5 rounded-full bg-emerald-400"></span><span>${apiConfig.model} (自訂金鑰已連線)</span>`;
            } else {
                indicator.className = 'w-2 h-2 rounded-full bg-blue-400';
                modeBadge.innerHTML = `<span class="w-1.5 h-1.5 rounded-full bg-blue-400"></span><span>${apiConfig.model} (內建環境 / 推演模式)</span>`;
            }

            if (groundingTag) {
                groundingTag.textContent = apiConfig.useGrounding ? 'SEARCH GROUNDING: ON' : 'SEARCH GROUNDING: OFF';
                groundingTag.className = apiConfig.useGrounding
                    ? 'text-[10px] px-2 py-0.5 rounded bg-[#131926] border border-[#1E293B] text-emerald-400 font-mono-num'
                    : 'text-[10px] px-2 py-0.5 rounded bg-[#131926] border border-[#1E293B] text-slate-400 font-mono-num';
            }
        }

        function showToast(msg, icon = '✅') {
            const box = document.getElementById('toast-box');
            document.getElementById('toast-text').textContent = msg;
            document.getElementById('toast-icon').textContent = icon;
            box.classList.remove('translate-y-12', 'opacity-0');
            setTimeout(() => box.classList.add('translate-y-12', 'opacity-0'), 3200);
        }

            renderICRoster();
            renderICArchiveList();
            refreshApiStatusIndicators();

            // Camera Reset & Depth of Field Toggle
            document.getElementById('btn-reset-cam').addEventListener('click', () => {
                camState.azimuth = Math.PI / 4;
                camState.polar = Math.PI / 3.1;
                camState.distance = 19.5;
                camState.target.set(0.4, 1.0, 0.2);
                updateCameraPosition();
                showToast('已重置為等角全景視角', '🎥');
            });

            const btnDof = document.getElementById('btn-toggle-dof');
            const viewport = document.getElementById('office-viewport');
            btnDof.addEventListener('click', () => {
                isDofActive = !isDofActive;
                viewport.classList.toggle('dof-active', isDofActive);
                btnDof.className = isDofActive
                    ? 'px-2.5 py-1.5 rounded-lg bg-[#F59E0B] text-slate-950 text-xs font-bold transition'
                    : 'px-2.5 py-1.5 rounded-lg bg-[#1E293B] hover:bg-slate-700 text-slate-300 text-xs font-medium transition';
                showToast(isDofActive ? '已開啟微縮模型景深模式' : '已關閉景深模式', '📷');
            });

            // Sidebar Tabs & Header History Shortcut
            const tabConsole = document.getElementById('tab-console');
            const tabArchive = document.getElementById('tab-archive');
            const panelConsole = document.getElementById('panel-console');
            const panelArchive = document.getElementById('panel-archive');

            function switchTab(target) {
                if (target === 'console') {
                    tabConsole.className = 'py-1.5 px-3 rounded-lg text-xs font-bold bg-[#1E293B] text-white shadow-sm transition flex items-center justify-center gap-1.5';
                    tabArchive.className = 'py-1.5 px-3 rounded-lg text-xs font-semibold text-slate-400 hover:text-slate-200 transition flex items-center justify-center gap-1.5';
                    panelConsole.classList.remove('hidden');
                    panelArchive.classList.add('hidden');
                } else {
                    tabArchive.className = 'py-1.5 px-3 rounded-lg text-xs font-bold bg-[#1E293B] text-white shadow-sm transition flex items-center justify-center gap-1.5';
                    tabConsole.className = 'py-1.5 px-3 rounded-lg text-xs font-semibold text-slate-400 hover:text-slate-200 transition flex items-center justify-center gap-1.5';
                    panelArchive.classList.remove('hidden');
                    panelConsole.classList.add('hidden');
                }
            }

            tabConsole.addEventListener('click', () => switchTab('console'));
            tabArchive.addEventListener('click', () => switchTab('archive'));
            document.getElementById('btn-header-history').addEventListener('click', () => switchTab('archive'));

            // Start IC Research Button & Preset Topics
            document.getElementById('btn-start-ic').addEventListener('click', () => {
                const input = document.getElementById('research-input');
                runInvestmentCommittee(input.value);
            });

            document.querySelectorAll('.preset-topic-btn').forEach(btn => {
                btn.addEventListener('click', () => {
                    const topic = btn.getAttribute('data-topic');
                    document.getElementById('research-input').value = topic;
                    runInvestmentCommittee(topic);
                });
            });

            // API Settings Modal Controls
            const apiModal = document.getElementById('api-settings-modal');
            document.getElementById('btn-open-api-modal').addEventListener('click', () => {
                document.getElementById('input-api-key').value = apiConfig.apiKey;
                document.getElementById('select-gemini-model').value = apiConfig.model;
                document.getElementById('checkbox-grounding').checked = apiConfig.useGrounding;
                apiModal.classList.remove('hidden');
            });

            const closeApiModal = () => apiModal.classList.add('hidden');
            document.getElementById('btn-close-api-modal').addEventListener('click', closeApiModal);
            document.getElementById('btn-cancel-api').addEventListener('click', closeApiModal);

            document.getElementById('btn-save-api').addEventListener('click', () => {
                apiConfig.apiKey = document.getElementById('input-api-key').value.trim();
                apiConfig.model = document.getElementById('select-gemini-model').value;
                apiConfig.useGrounding = document.getElementById('checkbox-grounding').checked;

                localStorage.setItem('ai_ic_gemini_key', apiConfig.apiKey);
                localStorage.setItem('ai_ic_gemini_model', apiConfig.model);
                localStorage.setItem('ai_ic_use_grounding', String(apiConfig.useGrounding));

                refreshApiStatusIndicators();
                closeApiModal();
                showToast('Gemini API 與聯網設定已儲存', '⚙️');
            });

            document.getElementById('btn-clear-api-key').addEventListener('click', () => {
                apiConfig.apiKey = '';
                localStorage.removeItem('ai_ic_gemini_key');
                document.getElementById('input-api-key').value = '';
                refreshApiStatusIndicators();
                showToast('已清除儲存的 API Key', '🗑️');
            });

            // IC Report Modal Close, Copy & Export
            const closeRepModal = () => document.getElementById('ic-report-modal').classList.add('hidden');
            document.getElementById('btn-close-report').addEventListener('click', closeRepModal);
            document.getElementById('btn-done-report').addEventListener('click', closeRepModal);
            document.getElementById('ic-report-modal').addEventListener('click', (e) => {
                if (e.target.id === 'ic-report-modal') closeRepModal();
            });

            document.getElementById('btn-copy-md').addEventListener('click', () => {
                if (!activeReport) return;
                const md = generateMarkdownFromReport(activeReport);
                const ta = document.createElement('textarea');
                ta.value = md;
                document.body.appendChild(ta);
                ta.select();
                document.execCommand('copy');
                document.body.removeChild(ta);
                showToast('已複製 Markdown 機構投研報告', '📋');
            });

            document.getElementById('btn-export-md').addEventListener('click', () => {
                if (!activeReport) return;
                const md = generateMarkdownFromReport(activeReport);
                const blob = new Blob([md], { type: 'text/markdown;charset=utf-8' });
                const url = URL.createObjectURL(blob);
                const a = document.createElement('a');
                a.href = url;
                a.download = `IC_Memo_${activeReport.topic.slice(0, 12)}.md`;
                a.click();
                URL.revokeObjectURL(url);
                showToast('已匯出投審會決議報告 (.md)', '⬇️');
            });
        });
